```bash
git clone git@github.com-luannt2002:luannt2002/agent-cli.git agent-cli   # hoặc HTTPS (đọc công khai)
cd agent-cli
git fetch origin 'refs/heads/*:refs/remotes/origin/*'
git switch -c v3-handoff origin/v3-handoff            # mã ứng viên + toàn bộ reports/
for b in v2-final v3-exposure v3-candidate; do git branch "$b" "origin/$b"; done
# baseline so theo tên (một lần):
git worktree add ../agent-cli-v2 v2-final
(cd ../agent-cli-v2 && GOMAXPROCS=2 go test -json -count=1 -p 2 ./... > ~/v2-suite.json)
# kiểm toàn bộ một commit (worktree riêng, không đụng cây đang làm):
SRC=$PWD BASE=~/v2-suite.json bash reports/v3-research/tools/verify-final.sh 300c8dd1 ~/v3ev/final/r4
```
